#include <WiFi.h>
#include <SPI.h>
#include <MFRC522.h>
#include <WebServer.h>
#include <Keypad.h>

// WiFi credentials
constexpr char ssid[] = "name";
constexpr char password[] = "password";

// RFID pins
#define RST_PIN 27
#define SS_PIN 22

// Relay
#define RELAY_PIN 5

// Keypad setup
const byte ROWS = 4, COLS = 3;
char keys[ROWS][COLS] = {
  {'1','2','3'},
  {'4','5','6'},
  {'7','8','9'},
  {'*','0','#'}
};
byte rowPins[ROWS] = {33, 32, 4, 26};
byte colPins[COLS] = {13, 12, 14};
Keypad keypad = Keypad(makeKeymap(keys), rowPins, colPins, ROWS, COLS);
char enteredPassword[10] = "";
constexpr char correctPassword[] = "0000";

// RFID
MFRC522 mfrc522(SS_PIN, RST_PIN);

// Web server
WebServer server(80);

// Allowed UIDs
const byte allowedUIDs[][4] = {
  // Replace the placeholders with actual card UIDs.
  {0xXX, 0xXX, 0xXX, 0xXX}, // Card 1: Replace with actual UID
  {0xXX, 0xXX, 0xXX, 0xXX}, // Card 2: Replace with actual UID
  {0xXX, 0xXX, 0xXX, 0xXX}, // Card 3: Replace with actual UID
  {0xXX, 0xXX, 0xXX, 0xXX}, // Card 4: Replace with actual UID
  {0xXX, 0xXX, 0xXX, 0xXX}, // Card 5: Replace with actual UID
  {0xXX, 0xXX, 0xXX, 0xXX}, // Card 6: Replace with actual UID
  {0xXX, 0xXX, 0xXX, 0xXX}, // Card 7: Replace with actual UID
  {0xXX, 0xXX, 0xXX, 0xXX}, // Card 8: Replace with actual UID
  {0xXX, 0xXX, 0xXX, 0xXX}  // Card 9: Replace with actual UID
};
constexpr size_t UID_COUNT = sizeof(allowedUIDs) / sizeof(allowedUIDs[0]);

void setup() {
  WiFi.mode(WIFI_STA);
  WiFi.setSleep(false);  // Disable WiFi sleep
  WiFi.setTxPower(WIFI_POWER_11dBm);  // You already have this

  SPI.begin(19, 25, 23); // SCK, MISO, MOSI
  //SPI.setFrequency(1000000);
  mfrc522.PCD_Init();

  pinMode(RELAY_PIN, OUTPUT);
  digitalWrite(RELAY_PIN, HIGH); // Relay off

  WiFi.begin(ssid, password);
  unsigned long startAttemptTime = millis();
  while (WiFi.status() != WL_CONNECTED && millis() - startAttemptTime < 10000) {
    delay(500);
  }

  server.on("/", HTTP_GET, []() {
    sendCORSHeaders();
    server.send(200, "text/plain", "ESP32 RFID Door Lock - Use /open to unlock.");
  });

  server.on("/open", HTTP_GET, []() {
    sendCORSHeaders();
    unlockDoor();
    server.send(200, "text/plain", "Door unlocked via Wi-Fi!");
  });

  server.begin();
}

void loop() {
  delay(10);  // Small delay for stability
  server.handleClient();
  checkRFID();
  checkKeypad();
}

void checkRFID() {
  if (mfrc522.PICC_IsNewCardPresent() && mfrc522.PICC_ReadCardSerial()) {
    if (isUIDAllowed(mfrc522.uid.uidByte)) {
      unlockDoor();
    }
    mfrc522.PICC_HaltA();
    mfrc522.PCD_StopCrypto1();
  }
}

void checkKeypad() {
  static byte index = 0;
  char key = keypad.getKey();

  if (!key) return;

  if (key == '#') {
    enteredPassword[index] = '\0'; // Null-terminate
    if (strcmp(enteredPassword, correctPassword) == 0) {
      unlockDoor();
    }
    index = 0;
    enteredPassword[0] = '\0';
  } else if (key == '*') {
    index = 0;
    enteredPassword[0] = '\0';
  } else if (index < sizeof(enteredPassword) - 1) {
    enteredPassword[index++] = key;
    enteredPassword[index] = '\0';
  }
}

void unlockDoor() {
  digitalWrite(RELAY_PIN, LOW);
  delay(400);
  digitalWrite(RELAY_PIN, HIGH);
  //here the only solution i found is to force the rstart of the esp so irt works every time cause the problem i got after teh use of the rfid "a problem due to the power efficiency"
  delay(200);
  ESP.restart();
}

bool isUIDAllowed(const byte* uid) {
  for (size_t i = 0; i < UID_COUNT; ++i) {
    if (memcmp(uid, allowedUIDs[i], 4) == 0) return true;
  }
  return false;
}

void sendCORSHeaders() {
  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.sendHeader("Access-Control-Allow-Methods", "GET");
}
